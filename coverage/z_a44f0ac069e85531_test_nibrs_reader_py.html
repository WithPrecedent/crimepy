40def _read(*args: Any, **kwargs: Any) -> dict[str, pd.DataFrame]: 

41 """Reads tables with `reader.read`, checking that they are datasets. 

42 

43 Args: 

44 *args: arguments for `reader.read`. 

45 **kwargs: arguments for `reader.read`. 

46 

47 Returns: 

48 The data frame of each dataset, by name. 

49 

50 """ 

51 datasets = reader.read(*args, **kwargs) 

52 assert all(isinstance(d, amos.Dataset) for d in datasets.values()) 

53 return {name: dataset.data for name, dataset in datasets.items()}

59def test_layouts_have_no_overlapping_fields() -> None: 

60 """Checks that no two fields of a layout share a position.""" 

61 for layout in layouts.LAYOUTS.values(): 

62 fields = sorted(layout, key = lambda f: f.start) 

63 for first, second in zip(fields, fields[1:], strict = False): 

64 assert first.end <= second.start, (layout.level, first, second)

67def test_layouts_have_unique_names_and_known_codebooks() -> None: 

68 """Checks that each layout's names are unique and its codebooks exist.""" 

69 for layout in layouts.LAYOUTS.values(): 

70 assert len(set(layout.names)) == len(layout.names) 

71 for field in layout: 

72 if field.codes is not None: 

73 assert field.codes in codes.CODEBOOKS
